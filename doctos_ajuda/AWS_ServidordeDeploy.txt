**** para iniciar como um serviço no servidor

criar um arquivo na app, raiz, carros_uwsgi.ini 

parâmetros

chdir           = /var/www/carros (diretório de onde está a aplicaçaõ)
module          = app.wsgi
home            = /var/www/carros/carrosv0 (ambiente virtual que será usado)
master          = true
processes       = 10 (quantidade de processos wsgi)
socket          = /var/www/carros/carros.sock
chmod-socket    = 666
vacuum          = true
